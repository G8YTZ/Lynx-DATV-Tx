LYNX-MIB DEFINITIONS ::= BEGIN

-- Lynx DATV: transmitter (Lynx-Tx), media player (Lynx Deck) and, later,
-- receiver (Lynx-Rx) status, for a repeater's monitoring (Home Assistant and
-- the like). One MIB for the family: the same file in each repository.
--
-- On Net-SNMP's experimental branch (netSnmpExperimental,
-- .1.3.6.1.4.1.8072.9999) until the repeater group has its own Private
-- Enterprise Number from IANA; then only the line "lynx ::= { ... }" and BASE
-- in the agents (Lynx-Tx's tools/lynx-snmp-agent.py, Lynx Deck's
-- tools/lynx-deck-snmp-agent.py) change. (Before 20 Nov 2026 this said
-- netSnmpPlaypen, .8072.9999.9999, which is not where the agents answer.)

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, Integer32, Gauge32, Counter32
        FROM SNMPv2-SMI
    DisplayString, TruthValue
        FROM SNMPv2-TC
    MODULE-COMPLIANCE, OBJECT-GROUP
        FROM SNMPv2-CONF
    netSnmpExperimental
        FROM NET-SNMP-MIB;

lynx MODULE-IDENTITY
    LAST-UPDATED "202611201600Z"
    ORGANIZATION "Lynx DATV"
    CONTACT-INFO "Justin Cockett, G8YTZ"
    DESCRIPTION  "Status of the Lynx DATV transmitter, media player and receiver."
    REVISION     "202611201600Z"
    DESCRIPTION  "Lynx Deck, the media player and scheduler (lynxDeck, 1 to 20);
                  lynx placed under netSnmpExperimental, where the agents
                  have always answered."
    REVISION     "202610231400Z"
    DESCRIPTION  "The ROCK's own health: fan, uptime, load, free space (15 to 18)."
    REVISION     "202610192000Z"
    DESCRIPTION  "First version: Lynx-Tx."
    ::= { netSnmpExperimental 1 }

lynxTx      OBJECT IDENTIFIER ::= { lynx 1 }
lynxRx      OBJECT IDENTIFIER ::= { lynx 2 }     -- reserved for Lynx-Rx
lynxDeck    OBJECT IDENTIFIER ::= { lynx 3 }
lynxConformance OBJECT IDENTIFIER ::= { lynx 9 }

lynxTxState OBJECT-TYPE
    SYNTAX      INTEGER { idle(1), onAir(2), mutedBySdr(3), fault(4), lockout(5) }
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION
        "idle: off air. onAir: transmitting. mutedBySdr: Lynx-Tx is on air
        but the SDR has turned its transmitter off by itself because its
        IQ stopped arriving (TX Off then TX On restores it). fault: the
        transmit chain stopped by itself, or Lynx-Tx is not answering.
        lockout: auto-recover restarted the transmitter too often (3 times
        in 60 minutes by default) and has given up until TX On is pressed."
    ::= { lynxTx 1 }

lynxRefState OBJECT-TYPE

    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION

        SDR runs on its own oscillator (still on air, slightly off
        frequency). unknown: the SDR cannot say (DL1JM's firmware)."
    ::= { lynxTx 3 }
